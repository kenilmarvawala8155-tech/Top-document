CREATE PROCEDURE CheckEmployeeDepartment(IN empId INT)
BEGIN
    DECLARE dept INT;

    SELECT dept_id INTO dept
    FROM employees
    WHERE emp_id = empId;

    IF dept = 1 THEN
        SELECT 'Employee belongs to HR department' AS message;
    ELSEIF dept = 2 THEN
        SELECT 'Employee belongs to IT department' AS message;
    ELSE
        SELECT 'Employee belongs to another department' AS message;

DELIMITER //
CREATE PROCEDURE DisplayEmployeeNames()
BEGIN
    DECLARE done INT DEFAULT 0;
    DECLARE empName VARCHAR(100);

    DECLARE emp_cursor CURSOR FOR
        SELECT emp_name FROM employees;

    DECLARE CONTINUE HANDLER FOR NOT FOUND SET done = 1;

    OPEN emp_cursor;

    read_loop: LOOP
        FETCH emp_cursor INTO empName;

        IF done THEN
            LEAVE read_loop;
        END IF;

        SELECT empName AS employee_name;
    END LOOP;

    CLOSE emp_cursor;
END //
DELIMITER ;
CALL DisplayEmployeeNames();


DELIMITER //

CREATE PROCEDURE DisplayEmployeeDetails()
BEGIN
    DECLARE done INT DEFAULT 0;
    DECLARE e_id INT;
    DECLARE e_name VARCHAR(100);
    DECLARE d_id INT;

    DECLARE emp_cursor CURSOR FOR
        SELECT emp_id, emp_name, dept_id FROM employees;

    DECLARE CONTINUE HANDLER FOR NOT FOUND SET done = 1;

    OPEN emp_cursor;

    read_loop: LOOP
        FETCH emp_cursor INTO e_id, e_name, d_id;

        IF done THEN
            LEAVE read_loop;
        END IF;

        SELECT e_id AS emp_id, e_name AS emp_name, d_id AS dept_id;
    END LOOP;

    CLOSE emp_cursor;
END //

DELIMITER ;
CALL DisplayEmployeeDetails();


DELIMITER //

CREATE PROCEDURE DisplayCourses()
BEGIN
    DECLARE done INT DEFAULT 0;
    DECLARE c_id INT;
    DECLARE c_name VARCHAR(100);
    DECLARE c_duration INT;

    DECLARE course_cursor CURSOR FOR
        SELECT course_id, course_name, course_duration FROM courses;

    DECLARE CONTINUE HANDLER FOR NOT FOUND SET done = 1;

    OPEN course_cursor;

    read_loop: LOOP
        FETCH course_cursor INTO c_id, c_name, c_duration;

        IF done THEN
            LEAVE read_loop;
        END IF;

        SELECT c_id AS course_id, c_name AS course_name, c_duration AS course_duration;
    END LOOP;

    CLOSE course_cursor;
END //

DELIMITER ;
CALL DisplayCourses();
